-- =====================================================
-- DATABASE PULSE (PostgreSQL)
-- =====================================================

-- =====================================================
-- UTENTI
-- =====================================================
CREATE TABLE IF NOT EXISTS users (
    id VARCHAR(32) PRIMARY KEY,
    username VARCHAR(32) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    email VARCHAR(100) UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =====================================================
-- PROFILO UTENTE
-- =====================================================
CREATE TABLE IF NOT EXISTS profile (
    user_id VARCHAR(32) PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE,
    bio VARCHAR(2000),
    avatar_path VARCHAR(4096),
    social_links VARCHAR(256)[],  -- array con max 8 link
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =====================================================
-- POST
-- =====================================================
CREATE TABLE IF NOT EXISTS posts (
    id SERIAL PRIMARY KEY,
    user_id VARCHAR(32) REFERENCES users(id) ON DELETE CASCADE,
    content VARCHAR(8192) NOT NULL,
    image_path VARCHAR(4096),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expires_at BIGINT NOT NULL  -- timestamp epoch
);

-- =====================================================
-- LIKES SUI POST
-- =====================================================
CREATE TABLE IF NOT EXISTS likes (
    user_id VARCHAR(32) REFERENCES users(id) ON DELETE CASCADE,
    post_id INT REFERENCES posts(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, post_id)
);

-- =====================================================
-- COMMENTI
-- =====================================================
CREATE TABLE IF NOT EXISTS comments (
    id SERIAL PRIMARY KEY,
    post_id INT REFERENCES posts(id) ON DELETE CASCADE,
    user_id VARCHAR(32) REFERENCES users(id) ON DELETE CASCADE,
    content VARCHAR(8192) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =====================================================
-- LIKE SUI COMMENTI
-- =====================================================
CREATE TABLE IF NOT EXISTS comment_likes (
    user_id VARCHAR(32) REFERENCES users(id) ON DELETE CASCADE,
    comment_id INT REFERENCES comments(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, comment_id)
);

-- =====================================================
-- LISTA PIN (utenti che metti in lista)
-- =====================================================
CREATE TABLE IF NOT EXISTS pins (
    owner_id VARCHAR(32) REFERENCES users(id) ON DELETE CASCADE,
    pinned_id VARCHAR(32) REFERENCES users(id) ON DELETE CASCADE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (owner_id, pinned_id)
);

CREATE TABLE IF NOT EXISTS post_views (
id BIGSERIAL PRIMARY KEY,
viewer_token TEXT NOT NULL,
post_id BIGINT NOT NULL,
created_at TIMESTAMPTZ DEFAULT NOW(),
UNIQUE (viewer_token, post_id)

/// VISUALIZZATO - MESSAGE TIME STUFF (VIEWS, DOT, SOCIAL ANXIETY STUFF)

CREATE TABLE conversations (
  id BIGSERIAL PRIMARY KEY,
  user_a UUID NOT NULL,
  user_b UUID NOT NULL,
  last_message TEXT,
  last_at TIMESTAMPTZ,
  unread_for_a INT DEFAULT 0,
  unread_for_b INT DEFAULT 0,
  user_min UUID GENERATED ALWAYS AS (LEAST(user_a, user_b)) STORED,
  user_max UUID GENERATED ALWAYS AS (GREATEST(user_a, user_b)) STORED,
  UNIQUE (user_min, user_max)
);



/// DIRTY OVERLAY
CREATE TABLE IF NOT EXISTS dirty (
    id BIGSERIAL PRIMARY KEY,
    user_id VARCHAR(32) UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    status VARCHAR(3) DEFAULT 'no' CHECK(status IN ('yes', 'no')),
    last_activity TIMESTAMPTZ DEFAULT NOW()
);
